sed -i "s|DEFAULT_OWNER_PASSWORD=.*|DEFAULT_OWNER_PASSWORD=${OWNER_PASSWORD}|" .env

# Set ownership BEFORE composer install so www-data can write
chown -R www-data:www-data ${INSTALL_PATH}

# Install Composer dependencies (as www-data)
log_info "Installing Composer dependencies..."
sudo -u www-data composer install --no-dev --optimize-autoloader --no-interaction --working-dir=${INSTALL_PATH}

# Install NPM dependencies and build assets using Bun (as www-data)
log_info "Installing NPM dependencies and building assets with Bun..."
export BUN_INSTALL="$HOME/.bun"
export PATH="$BUN_INSTALL/bin:$PATH"
cd ${INSTALL_PATH}
sudo -E -u www-data bash -c "export PATH=$BUN_INSTALL/bin:\$PATH && cd ${INSTALL_PATH} && bun install && bun run build"

# If bun as www-data fails, fallback to running as root then fixing permissions
if [[ $? -ne 0 ]]; then
    log_warn "Bun as www-data failed, running as root and fixing permissions..."
    cd ${INSTALL_PATH}
    bun install
    bun run build
    chown -R www-data:www-data ${INSTALL_PATH}
fi

# Set permissions

configure_supervisor() {
log_info "Configurando Supervisor para trabajadores de cola y chat..."

1
cat > /etc/supervisor/conf.d/unit3d.conf << EOF
[program:unit3d-queue]
process_name=%(program_name)s_%(process_num)02d
command=php ${INSTALL_PATH}/artisan queue:work redis --sleep=3 --tries=3 --max-time=3600
autostart=true
autorestart=true
stopasgroup=true
killasgroup=true
user=www-data
numprocs=4
redirect_stderr=true
stdout_logfile=${INSTALL_PATH}/storage/logs/queue.log
stopwaitsecs=3600

[programa:unit3d-echo]
process_name=%(program_name)s_%(process_num)02d
command=laravel-echo-server start --dir=${INSTALL_PATH}
autostart=true
autorestart=true
stopasgroup=true
killasgroup=true
user=www-data
numprocs=1
redirect_stderr=true
stdout_logfile=${INSTALL_PATH}/storage/logs/echo.log
EOF

1 
2 
3 
4 
5
supervisorctl relee
Actualización de supervisorctl
supervisorctl iniciar todo

log_info "Supervisor configurado correctamente"
}
